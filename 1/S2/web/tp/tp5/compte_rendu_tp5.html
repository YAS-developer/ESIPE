Exercice 1 - Stock Tickers

  1. Créer le bouton "Refresh" et le champ texte dans le fichier HTML

              <div>
            <button id="refreshButton">Refresh</button>
            <input type="text" id="markText" placeholder="Enter threshold value">
          </div>
          <table id="stockTable"></table>

  2. Configurer le serveur Java

    Exécutez le fichier JExpress.java avec la commande : 

              java JExpress.java

  3. Implémenter le script JavaScript

              document.getElementById('refreshButton').onclick = function() {
            fetch('/api/ticker')
              .then(response => response.json())
              .then(tickers => {
                console.log(tickers);
                // Pour l'instant, affichons simplement les tickers
                populateTable(tickers);
              });
          };

          function populateTable(tickers) {
            const table = document.getElementById('stockTable');
            table.innerHTML = ''; // Réinitialiser la table
            tickers.forEach(ticker => {
              const row = document.createElement('tr');
              const cell = document.createElement('td');
              cell.textContent = ticker;
              row.appendChild(cell);
              table.appendChild(row);
            });
          }

  4. Explication de async et await

    Les mots-clés async et await simplifient la gestion des promesses en JavaScript. async permet de déclarer une fonction qui retourne une promesse, et await permet d'attendre la résolution d'une promesse de manière asynchrone. 

              document.getElementById('refreshButton').onclick = async function() {
            try {
              const response = await fetch('/api/ticker');
              const tickers = await response.json();
              console.log(tickers);
              populateTable(tickers);
            } catch (error) {
              console.error('Error fetching tickers:', error);
            }
          };

  5. Ajouter les quotations à la table

              async function fetchTickers() {
            const response = await fetch('/api/ticker');
            return response.json();
          }

          async function fetchStocks() {
            const response = await fetch('/api/stock', { method: 'POST' });
            return response.json();
          }

          async function refreshData() {
            try {
              const tickers = await fetchTickers();
              const stocks = await fetchStocks();
              const stockTable = document.getElementById('stockTable');
              stockTable.innerHTML = '';
              tickers.forEach(ticker => {
                const row = document.createElement('tr');
                const tickerCell = document.createElement('td');
                const stockCell = document.createElement('td');
                tickerCell.textContent = ticker;
                stockCell.textContent = stocks[ticker] || '-';
                row.appendChild(tickerCell);
                row.appendChild(stockCell);
                stockTable.appendChild(row);
              });
            } catch (error) {
              console.error('Error:', error);
            }
          }

          document.getElementById('refreshButton').onclick = refreshData;

  6. Affichage des quotations en fonction du seuil

              function highlightStocks(threshold) {
            const rows = document.querySelectorAll('#stockTable tr');
            rows.forEach(row => {
              const value = parseInt(row.children[1].textContent, 10);
              if (!isNaN(value) && value < threshold) {
                row.children[1].style.color = 'green';
              } else {
                row.children[1].style.color = 'black';
              }
            });
          }

          document.getElementById('markText').onkeyup = function() {
            const threshold = parseInt(this.value, 10);
            if (!isNaN(threshold)) {
              highlightStocks(threshold);
            }
          };

  7. Gérer les requêtes multiples

              let refreshTimeout;

          async function refreshData() {
            clearTimeout(refreshTimeout); // Clear previous timeout to avoid multiple triggers
            try {
              const tickers = await fetchTickers();
              const stocks = await fetchStocks();
              const stockTable = document.getElementById('stockTable');
              stockTable.innerHTML = '';
              tickers.forEach(ticker => {
                const row = document.createElement('tr');
                const tickerCell = document.createElement('td');
                const stockCell = document.createElement('td');
                tickerCell.textContent = ticker;
                stockCell.textContent = stocks[ticker] || '-';
                row.appendChild(tickerCell);
                row.appendChild(stockCell);
                stockTable.appendChild(row);
              });
              const threshold = parseInt(document.getElementById('markText').value, 10);
              if (!isNaN(threshold)) {
                highlightStocks(threshold);
              }
              refreshTimeout = setTimeout(refreshData, 1000);
            } catch (error) {
              console.error('Error:', error);
            }
          }

          document.getElementById('refreshButton').onclick = refreshData;

  8. Pourquoi utiliser POST pour /api/stock

    Une requête POST a été choisie pour /api/stock car elle permet de modifier l'état du serveur ou de mettre à jour les ressources, contrairement à GET qui est censé être utilisé uniquement pour obtenir des données sans effets secondaires.
